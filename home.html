# Terraform Production Blueprint — Hybrid Cloud & AKS Cluster
terraform {
  required_version = ">= 1.7.0"
  backend "azurerm" {
    resource_group_name  = "rg-tfstate-prod-sea"
    storage_account_name = "sttfstatearunprod"
    container_name       = "tfstate"
    key                  = "prod.infrastructure.tfstate"
  }
}

module "network_spoke" {
  source              = "./modules/azure-vnet-spoke"
  environment         = "production"
  location            = "southeastasia"
  address_space       = ["10.40.0.0/16"]
  enable_ddos_protect = true
}

module "aks_production" {
  source              = "./modules/aks-enterprise"
  cluster_name        = "aks-rws-prod-sea-01"
  kubernetes_version  = "1.30"
  vnet_subnet_id      = module.network_spoke.aks_subnet_id
  enable_auto_scaling = true
  min_count           = 3
  max_count           = 12
  tags = {
    ManagedBy = "Terraform"
    Owner     = "Arun-Kumar-IaC"
    SLA       = "99.99-HA"
  }
}
